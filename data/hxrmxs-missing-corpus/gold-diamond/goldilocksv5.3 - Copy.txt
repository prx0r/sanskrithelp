async def discover_kg_patterns(self) -> List[Dict]:
    interactions = await self.postgres.fetch(
        "SELECT user_message, ai_response, framework_id FROM interactions WHERE helpfulness_rating >= 4"
    )
    texts = [f"{i['user_message']} {i['ai_response']}" for i in interactions]
    topics = self.bertopic_model.fit_transform(texts)[0]
    
    hypotheses = []
    for topic_id, framework_id in zip(topics, [i["framework_id"] for i in interactions]):
        if topic_id != -1:
            hypotheses.append({
                "source_framework": framework_id,
                "target_framework": f"topic_{topic_id}",
                "confidence": 0.8,
                "supporting_interactions": len([t for t in topics if t == topic_id])
            })
    
    await self.postgres.execute(
        "INSERT INTO knowledge_graph_hypotheses (hypothesis_id, data) VALUES ($1, $2)",
        str(uuid.uuid4()), json.dumps(hypotheses)
    )
    return hypotheses

async def analyze_rct_outcomes(self) -> Dict:
    outcomes = await self.postgres.fetch(
        "SELECT uea.experimental_group, co.phq9_score FROM clinical_outcomes co JOIN user_experimental_assignments uea ON co.user_id = uea.user_id"
    )
    df = pd.DataFrame(outcomes)
    stats = {}
    for group in df["experimental_group"].unique():
        group_data = df[df["experimental_group"] == group]["phq9_score"]
        t_stat, p_value = ttest_rel(group_data, df[df["experimental_group"] == "CONTROL"]["phq9_score"])
        stats[group] = {"p_value": p_value, "mean_score": group_data.mean()}
    return stats

async def forward(self, text: str, context: str = "") -> Dict[str, float]:
    input_text = f"{context} {text}".strip()
    text_embedding = self.embedder.encode(input_text)
    crisis_similarities = [
        np.dot(text_embedding, kw_emb) / (np.linalg.norm(text_embedding) * np.linalg.norm(kw_emb))
        for kw_emb in self.crisis_keyword_embeddings
    ]
    scores = self._compute_classifier_scores(input_text)
    scores["crisis"] = min(1.0, scores["crisis"] + max(crisis_similarities) * 0.2)
    return scores

1. Executive Summary
This document outlines the final, synthesized development plan for the HelloKoa project. This plan represents the integration of the most powerful concepts from all previous architectural iterations, creating a cohesive and actionable roadmap.

2. Core Data Assets: The Foundation of HelloKoa's Intelligence
HelloKoa's unique capabilities are derived from two foundational data sources. Understanding their role is essential to understanding the architecture.

A. jsontests2.txt: The Therapeutic Component Library (TCL)
This file represents the Knowledge Core of the AI. It is not simply training data; it's a structured, evidence-based library of discrete therapeutic skills.

Structure: Each component in the TCL is a JSON object with a clear schema, including a component_id, pillar (e.g., "Deepening Empathic Understanding"), theory (e.g., "person_centered"), a skill_definition, and an execution_process.


Key Feature: learning_options: Crucially, each component contains a set of learning_options. These provide multiple, expert-written examples of how to apply the skill correctly in a given situation. They also include examples flagged for removal ("safety_flag": "REMOVE_FROM_TRAINING") to prevent the AI from learning anti-patterns.

Function: The TCL serves as the AI's "textbook." The scripts/ingest_tcl.py script parses this file to populate the application's database, and the scripts/generate_tcl_training_data.py script uses the learning_options to create a rich, situational training set for the AI's initial training.
B. truenorfjsons.txt: The Golden Standard Dialogues
This file represents the Benchmark for Therapeutic Wisdom. It contains transcripts of expert human therapists conducting sessions.

Structure: Each entry is a complete dialogue with multiple turns between a "client" and a "therapist". Each dialogue is tagged with a therapeutic_principle that describes the expert's strategy (e.g., "Navigates the therapeutic process by addressing silence and fear...").
Function: This file is not used for initial training. Instead, it is the cornerstone of the Golden Standard Refinement Loop. The scripts/validate_with_golden_standards.py script  uses these dialogues to test the live AI. It compares the AI's response to the expert's response and uses the critique_and_correct_prompt.txt  to generate "causal fix" training data. This teaches the AI to close the gap between its own performance and that of a human expert, allowing it to learn nuance and wisdom over time.
